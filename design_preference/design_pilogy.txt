  function calculateQuickMatrix() {
    const d = document.getElementById('dobDay').value;
    const m = document.getElementById('dobMonth').value;
    const y = document.getElementById('dobYear').value;
    const vibeCard = document.getElementById('vibeCard');
    const numeralEl = document.getElementById('lifePathNumeral');
    const previewText = document.getElementById('lifePathPreviewText');

    if (!d || !m || !y) {
      vibeCard.classList.add('hidden');
      return;
    }

    // Pythagoras Single Digit Life Path Calculation (with master numbers 11, 22)
    const digits = (d + m + y).split('').map(Number);
    let total = digits.reduce((acc, curr) => acc + curr, 0);

    function reduceNum(n) {
      if (n === 11 || n === 22 || n <= 9) return n;
      const sum = String(n).split('').map(Number).reduce((a, b) => a + b, 0);
      return sum > 9 && sum !== 11 && sum !== 22 ? reduceNum(sum) : sum;
    }

    const lifePath = reduceNum(total);

    const descriptions = {
      2: 'Sứ giả hòa bình & thấu cảm',
      3: 'Năng lượng sáng tạo & ngôn từ',
      4: 'Cấu trúc vững chãi & thực tiễn',
      5: 'Tự do khai phóng & trải nghiệm',
      6: 'Trái tim bảo bọc & tình thương',
      7: 'Trực giác triết học & chiêm nghiệm',
      8: 'Bản lĩnh độc lập & kiến thiết',
      9: 'Nhân văn vị tha & lý tưởng cao',
      10: 'Khả năng thích ứng & linh hoạt cao',
      11: 'Bậc thầy trực giác tâm linh',
      22: 'Kiến trúc sư của tương lai'
    };

    numeralEl.textContent = lifePath;
    previewText.textContent = descriptions[lifePath] || 'Đường đời rung động cao';
    vibeCard.classList.remove('hidden');
  }

      if (!toast) return;
      toastText.textContent = msg;
      toast.classList.remove('opacity-0', 'pointer-events-none');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        toast.classList.add('opacity-0', 'pointer-events-none');

          cacheStatusText.classList.add('text-secondary');

      if (!dangerModal) return;

        dangerModal.classList.add('opacity-0', 'pointer-events-none');
      }

    if (openDangerModalBtn) openDangerModalBtn.addEventListener('click', () => toggleModal(true));
